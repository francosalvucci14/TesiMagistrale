\begin{frame}[t]{Neighborhood Search statico: intersezione e Jaccard}
\vspace{-.16cm}

    \item \textbf{Query:} $S\subseteq V$.\\[4pt]
    \item \textbf{Goal:} trovare $u\in V$ tale che $\sigma(S,N(u))\geq\tau$.
